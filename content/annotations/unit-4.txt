formula: R_n(x) = f(x) - P_n(x)

formula: e^x = \sum_{k=0}^n \frac{x^k}{k!} + R_n(x) = 1 + x + \frac{x^2}{2!} + \dots + \frac{x^n}{n!} + R_n(x)
formula: \sin x = \sum_{k=0}^n \frac{(-1)^k}{(2k+1)!} x^{2k+1} + R_{2n+2}(x) = x - \frac{x^3}{3!} + \frac{x^5}{5!} - \dots + \frac{(-1)^n}{(2n+1)!} x^{2n+1} + R_{2n+2}(x)
